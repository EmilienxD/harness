// Lifecycle and concurrent nested-tool orchestration only; production owns report behavior.
import { Type } from "typebox";
import { writeFile } from "node:fs/promises";
import { join } from "node:path";
export default function(pi) {
  pi.registerCommand("gotchas-reload", { handler: async (_, ctx) => ctx.reload() });
  pi.registerCommand("gotchas-anchor", { handler: () => pi.appendEntry("gotchas-anchor", {}) });
  pi.registerCommand("gotchas-back", { handler: async (_, ctx) => {
    const anchor = ctx.sessionManager.getEntries().find(e => e.type === "custom" && e.customType === "gotchas-anchor");
    await ctx.navigateTree(anchor.id, { summarize: false });
  } });
  pi.registerCommand("gotchas-fork", { handler: async (_, ctx) => {
    const user = ctx.sessionManager.getBranch().find(e => e.type === "message" && e.message.role === "user");
    await ctx.fork(user.id);
  } });
  pi.registerTool({ name: "gotchas_nested", label: "Nested gotchas fixture", description: "Exercise concurrent nested submissions", parameters: Type.Object({}),
    async execute(_, args, signal, onUpdate, ctx) {
      const message = "**Concurrent nested report**\n\n- One consequential caveat.";
      const results = await Promise.all([ctx.executeTool("report_gotchas", { message }, { signal }), ctx.executeTool("report_gotchas", { message }, { signal })]);
      await writeFile(join(ctx.cwd, "nested-evidence.json"), JSON.stringify(results.map(r => r.result.structuredContent)));
      return { content: [{ type: "text", text: "Nested submission completed." }], details: undefined };
    },
  });
}
