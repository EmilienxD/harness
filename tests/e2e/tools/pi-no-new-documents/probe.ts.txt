// Test-only lifecycle and nested-tool orchestration; shipped extension owns decisions.
import { writeFile } from "node:fs/promises";
import { join } from "node:path";
import { Type } from "typebox";

export default function (pi: any) {
  pi.registerTool({
    name: "documents_probe", label: "Documents probe", description: "Exercise nested document writes",
    parameters: Type.Object({}),
    execute: async (_id: string, _args: unknown, _signal: unknown, _update: unknown, ctx: any) => {
      const denied = await ctx.executeTool("write", { path: "@docs/nested/new.md", content: "forbidden\n" });
      const draft = await ctx.executeTool("write", { path: ".harness/doc-drafts/docs/nested/new.md", content: "nested content\n" });
      await writeFile(join(ctx.cwd, "nested-result.json"), JSON.stringify({ denied: denied.isError, draft: draft.isError }));
      return { content: [{ type: "text", text: "Nested probe complete." }] };
    },
  });
  pi.registerCommand("documents-anchor", { handler: () => pi.appendEntry("documents-anchor", {}) });
  pi.registerCommand("documents-auto-approve", { handler: () => pi.sendUserMessage("/approve-doc-draft docs/auto.md") });
  pi.registerCommand("documents-dispatch-approve", { handler: () => pi.sendUserMessage("/approve-doc-draft docs/dispatch.md", { expandPromptTemplates: true }) });
  pi.registerCommand("documents-reload", { handler: async (_: string, ctx: any) => ctx.reload() });
  pi.registerCommand("documents-inspect", { handler: async (_: string, ctx: any) => {
    const entries = ctx.sessionManager.getBranch();
    await writeFile(join(ctx.cwd, "notice-count.json"), JSON.stringify({
      outcomes: entries.filter((e: any) => e.type === "custom_message" && e.customType === "harness-document-drafts" && Object.keys(e.details?.outcomes ?? {}).length).length,
      pending: entries.filter((e: any) => e.type === "custom_message" && e.customType === "harness-pending-document-drafts").length,
    }));
  } });
  pi.registerCommand("documents-back", { handler: async (_: string, ctx: any) => {
    const anchor = ctx.sessionManager.getBranch().find((e: any) => e.type === "custom" && e.customType === "documents-anchor");
    await ctx.navigateTree(anchor.id, { summarize: false });
  } });
}
