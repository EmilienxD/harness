// Test-only extension proving native project discovery and lifecycle. No production policy lives here.
import { appendFile, writeFile } from "node:fs/promises";
import { basename, join } from "node:path";
import type { ExtensionAPI } from "@earendil-works/pi-coding-agent";

export default function (pi: ExtensionAPI) {
  const record = (cwd: string, value: unknown) => appendFile(join(cwd, "probe-events.jsonl"), `${JSON.stringify(value)}\n`, "utf8");
  pi.on("session_start", async (event, ctx) => { await record(ctx.cwd, { type: event.type, reason: event.reason }); });
  pi.on("session_shutdown", async (event, ctx) => { await record(ctx.cwd, { type: event.type, reason: event.reason }); });
  pi.on("tool_call", async (event, ctx) => {
    await record(ctx.cwd, { type: event.type, name: event.toolName });
    if (event.toolName === "write" && typeof event.input.path === "string" && basename(event.input.path) === "blocked.txt") return { block: true, reason: "Harness fixture denies this write" };
    return undefined;
  });
  pi.on("tool_result", async (event, ctx) => { await record(ctx.cwd, { type: event.type, name: event.toolName, isError: event.isError }); });
  pi.on("agent_settled", async (event, ctx) => { await record(ctx.cwd, { type: event.type, aborted: event.aborted }); });
  pi.registerCommand("e2e-touch", {
    description: "Write the test fixture's deterministic command artifact",
    handler: async (_args, ctx) => { await writeFile(join(ctx.cwd, "command-result.txt"), "command\n", "utf8"); pi.appendEntry("harness-e2e-command", { completed: true }); },
  });
  pi.registerCommand("e2e-reload", {
    description: "Reload the real Pi extension runtime",
    handler: async (_args, ctx) => { await ctx.reload(); },
  });
}
